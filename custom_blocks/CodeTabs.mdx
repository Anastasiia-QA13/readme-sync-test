import { useState } from "react";

export default function CodeTabs({
  tabs = {
    curl: `curl -H "Authorization: Bearer $API_KEY" https://api.example.com/ping`,
    javascript: `await fetch("https://api.example.com/ping", { headers: { Authorization: "Bearer " + apiKey } })`,
    python: `import requests\nrequests.get("https://api.example.com/ping", headers={"Authorization": f"Bearer {api_key}"})`
  }
}) {
  const [active, setActive] = useState(Object.keys(tabs)[0]);
  return (
    <div style={{ border: "1px solid #e5e7eb", borderRadius: 8, overflow: "hidden", fontFamily: "Inter, sans-serif" }}>
      <div style={{ display: "flex", gap: 8, padding: 8, background: "#f8fafc", borderBottom: "1px solid #e5e7eb" }}>
        {Object.keys(tabs).map(k => (
          <button
            key={k}
            onClick={() => setActive(k)}
            style={{
              padding: "6px 10px",
              borderRadius: 6,
              border: "1px solid #e5e7eb",
              background: active === k ? "#111827" : "white",
              color: active === k ? "white" : "#111827",
              cursor: "pointer",
              textTransform: "capitalize"
            }}
          >
            {k}
          </button>
        ))}
      </div>
      <pre style={{ margin: 0, padding: "12px 14px", background: "#0b1220", color: "white", overflowX: "auto" }}>
{tabs[active]}
      </pre>
    </div>
  );
}
